Sincronización en Pareja (Google Sheets)

Al guardar un movimiento se envía automáticamente a Google Sheets. Pulsa Sincronizar ahora para descargar los registros nuevos de tu pareja.

Última sincronización: Sin sincronizar aún
Instrucciones (2 minutos):
  1. Crea una hoja nueva en Google Sheets y ve a Extensiones > Apps Script.
  2. Borra el código que aparece y pega el script de abajo. Guarda el proyecto.
  3. Pulsa Implementar > Nueva implementación > Tipo: Aplicación web.
  4. En "Quién tiene acceso" elige Cualquier usuario, implementa y copia la URL resultante aquí arriba (en tu móvil y en el de tu pareja).
function doGet(e) {
  var sheet = getOrCreateSheet();
  var data = sheet.getDataRange().getValues();
  var tz = Session.getScriptTimeZone();
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    if (!data[i][0]) continue;
    var rawDate = data[i][3];
    var formattedDate = (rawDate instanceof Date)
      ? Utilities.formatDate(rawDate, tz, 'yyyy-MM-dd')
      : String(rawDate).slice(0, 10);
    rows.push({
      id: String(data[i][0]),
      type: String(data[i][1]),
      amount: Number(data[i][2]),
      date: formattedDate,
      category: String(data[i][4]),
      method: String(data[i][5]),
      person: String(data[i][6]),
      isFixed: data[i][7] === true || data[i][7] === 'TRUE',
      description: String(data[i][8] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', movements: rows }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = getOrCreateSheet();
  var payload = JSON.parse(e.postData.contents);
  var action = payload.action;

  if (action === 'upsert') {
    var m = payload.movement;
    var data = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(m.id)) {
        rowIndex = i + 1;
        break;
      }
    }
    var rowValues = [m.id, m.type, m.amount, m.date, m.category, m.method, m.person, m.isFixed, m.description || ''];
    if (rowIndex > 0) {
      sheet.getRange(rowIndex, 1, 1, rowValues.length).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }
  } else if (action === 'delete') {
    var delId = String(payload.id);
    var dataDel = sheet.getDataRange().getValues();
    for (var j = dataDel.length - 1; j >= 1; j--) {
      if (String(dataDel[j][0]) === delId) {
        sheet.deleteRow(j + 1);
      }
    }
  } else if (action === 'bulkSync') {
    var list = payload.movements || [];
    sheet.clearContents();
    sheet.appendRow(['id', 'type', 'amount', 'date', 'category', 'method', 'person', 'isFixed', 'description']);
    list.forEach(function(m) {
      sheet.appendRow([m.id, m.type, m.amount, m.date, m.category, m.method, m.person, m.isFixed, m.description || '']);
    });
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Movimientos');
  if (!sheet) {
    sheet = ss.insertSheet('Movimientos');
    sheet.appendRow(['id', 'type', 'amount', 'date', 'category', 'method', 'person', 'isFixed', 'description']);
  }
  return sheet;
}
